-- DROP FUNCTION archis_meldingen.update_display_coordinates();

CREATE OR REPLACE FUNCTION archis_meldingen.update_display_coordinates()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
                BEGIN
                    NEW.display_coordinates = (
                        SELECT COALESCE(jsonb_agg(ST_AsGeoJSON(transformed_geom)::jsonb), '[]'::jsonb)
                        FROM (
                            SELECT 
                                CASE 
                                    WHEN NEW.crs = '3857' 
                                    THEN ST_Transform(ST_SetSRID(ST_Centroid(geom), 3857), 4326) 
                                    ELSE ST_Centroid(geom) 
                                END AS transformed_geom
                            FROM LATERAL (SELECT (ST_Dump(NEW.geometry)).geom) AS geo
                        ) AS processed
                    );
                    RETURN NEW;
                END;
                $function$
;
